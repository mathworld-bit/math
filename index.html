function setRandomWhatsAppLink() { const btn = document.getElementById('whatsapp-btn'); // أرقام الهواتف const myNumber = "971556595131"; const friendNumber = "971568245144"; // 1. توليد 5 أرقام عشوائية (مثلاً بين 1 و 100) let randomNumbers = []; for (let i = 0; i < 5; i++) { randomNumbers.push(Math.floor(Math.random() * 100) + 1); } // 2. إيجاد مجموع الأرقام الخمسة const sum = randomNumbers.reduce((acc, curr) => acc + curr, 0); // 3. التحقق من الناتج (زوجي أم فردي) وتحديد الرقم والاسم let finalNumber; let tutorName; if (sum % 2 === 0) { // الناتج زوجي (Even) -> رقمك أنت finalNumber = myNumber; tutorName = "حمزة"; } else { // الناتج فردي (Odd) -> رقم صديقك finalNumber = friendNumber; tutorName = "أحمد"; // يمكنك تغيير الاسم لاسم صديقك الحقيقي } // الرسالة المجهزة مسبقاً const message = encodeURIComponent(`مرحباً أستاذ ${tutorName}، أود الاستفسار عن دروس الرياضيات.`); // 4. تحديث الرابط والنص في الصفحة btn.href = `https://wa.me/${finalNumber}?text=${message}`; btn.innerText = `تواصل مع الأستاذ ${tutorName}`; // (اختياري للتدقيق) طباعة الأرقام والمجموع في الـ Console console.log("الأرقام العشوائية:", randomNumbers); console.log("المجموع:", sum); console.log("النتيجة:", sum % 2 === 0 ? "زوجي (رقمك)" : "فردي (صديقك)"); } // تشغيل الدالة عند تحميل الصفحة window.onload = setRandomWhatsAppLink;
